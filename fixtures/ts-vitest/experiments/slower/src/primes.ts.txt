// Number of primes below n, by trial division over every divisor.
export function countPrimes(n: number): number {
  let count = 0;
  for (let k = 2; k < n; k++) {
    let prime = true;
    for (let d = 2; d * d <= k; d++) {
      if (k % d === 0) {
        prime = false;
        break;
      }
    }
    if (prime) count++;
  }
  return count;
}
