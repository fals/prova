// Number of primes below n, by the sieve of Eratosthenes.
export function countPrimes(n: number): number {
  if (n <= 2) return 0;
  const composite = new Uint8Array(n);
  let count = 0;
  for (let i = 2; i < n; i++) {
    if (composite[i]) continue;
    count++;
    for (let j = i * i; j < n; j += i) composite[j] = 1;
  }
  return count;
}
