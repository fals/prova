namespace Calc;

public static class Primes
{
    /// <summary>Counts primes below n with a sieve of Eratosthenes.</summary>
    public static int CountPrimes(int n)
    {
        if (n < 3)
        {
            return 0;
        }
        var composite = new bool[n];
        int count = 0;
        for (int i = 2; i < n; i++)
        {
            if (!composite[i])
            {
                count++;
                for (long j = (long)i * i; j < n; j += i)
                {
                    composite[j] = true;
                }
            }
        }
        return count;
    }
}
