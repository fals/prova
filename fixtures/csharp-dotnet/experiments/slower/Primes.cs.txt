namespace Calc;

public static class Primes
{
    /// <summary>Counts primes below n by trial division over every divisor (no even skipping).</summary>
    public static int CountPrimes(int n)
    {
        int count = 0;
        for (int k = 2; k < n; k++)
        {
            if (IsPrime(k))
            {
                count++;
            }
        }
        return count;
    }

    private static bool IsPrime(int k)
    {
        if (k < 2)
        {
            return false;
        }
        for (int d = 2; d * d <= k; d++)
        {
            if (k % d == 0)
            {
                return false;
            }
        }
        return true;
    }
}
