def count_primes(n):
    """Number of primes below n, by trial division over every divisor."""
    count = 0
    for k in range(2, n):
        d = 2
        prime = True
        while d * d <= k:
            if k % d == 0:
                prime = False
                break
            d += 1
        if prime:
            count += 1
    return count
