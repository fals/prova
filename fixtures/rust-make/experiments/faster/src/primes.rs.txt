/// Number of primes below `n` (sieve of Eratosthenes).
pub fn count_primes(n: u32) -> u32 {
    let n = n as usize;
    if n < 3 {
        return 0;
    }
    let mut composite = vec![false; n];
    let mut count = 0;
    for k in 2..n {
        if composite[k] {
            continue;
        }
        count += 1;
        let mut m = k * k;
        while m < n {
            composite[m] = true;
            m += k;
        }
    }
    count
}
