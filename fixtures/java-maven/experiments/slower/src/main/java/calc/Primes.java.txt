package calc;

public final class Primes {
    private Primes() {}

    /** Counts primes below n by trial division over every divisor (no even skipping). */
    public static int countPrimes(int n) {
        int count = 0;
        for (int k = 2; k < n; k++) {
            if (isPrime(k)) {
                count++;
            }
        }
        return count;
    }

    private static boolean isPrime(int k) {
        if (k < 2) {
            return false;
        }
        for (int d = 2; d * d <= k; d++) {
            if (k % d == 0) {
                return false;
            }
        }
        return true;
    }
}
